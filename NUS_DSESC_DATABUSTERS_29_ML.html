# Load required packages
library(e1071)

# Create a data frame from the GDPGrowth xts object.

df <- data.frame(date = index(GDPGrowth["1992/2024"]),
                 GDPGrowth = as.numeric(GDPGrowth["1992/2024"]))


# Define the number of lags to use as predictors.
lag_order <- 4

# Create lagged predictors using dplyr::lag to ensure proper lengths.
for(i in 1:lag_order) {
  df[[paste0("lag", i)]] <- dplyr::lag(df$GDPGrowth, n = i)
}

# Remove rows with NA values (these arise from lagging).
df_svr <- na.omit(df)
df_svr$date <- as.Date(df_svr$date)

# Split the data into training and testing sets.
train_svr <- df_svr %>% filter(date < as.Date("2011-01-01"))
test_svr  <- df_svr %>% filter(date >= as.Date("2011-01-01"))

# Fit the SVR model using the lagged predictors.
svr_model <- svm(GDPGrowth ~ lag1 + lag2 + lag3 + lag4, 
                 data = train_svr, 
                 type = "eps-regression")

# Predict on the test set.
svr_predictions <- predict(svr_model, newdata = test_svr)

# Calculate forecast errors and RMSFE.
errors_svr <- test_svr$GDPGrowth - svr_predictions
rmsfe_svr <- sqrt(mean(errors_svr^2, na.rm = TRUE))
print(paste("SVR RMSFE:", rmsfe_svr))

# Load required libraries
library(e1071)
library(dplyr)
library(xts)
library(zoo)
library(ggplot2)

# ---------------------------
# 1. Prepare the Data for SVR
# ---------------------------
# Assume GDPGrowth is an xts object with quarterly GDP growth values up to 2024 Q4.
# Convert the xts object to a data frame.
df <- data.frame(date = index(GDPGrowth),
                 GDPGrowth = as.numeric(GDPGrowth))
df$date <- as.Date(df$date)

# Define the number of lags to use as predictors.
lag_order <- 4

# Create lagged predictors. (Using dplyr::lag to preserve the order.)
for(i in 1:lag_order) {
  df[[paste0("lag", i)]] <- dplyr::lag(df$GDPGrowth, n = i)
}

# Remove rows with NA values (introduced by lagging).
df_svr <- na.omit(df)

# ---------------------------
# 2. Fit the SVR Model
# ---------------------------
# Use the entire available historical data (df_svr) as training data.
# (You can adjust the training period if needed.)
svr_model <- svm(GDPGrowth ~ lag1 + lag2 + lag3 + lag4, 
                 data = df_svr, 
                 type = "eps-regression")

# ---------------------------
# 3. Forecast 2025 Recursively Using the SVR Model
# ---------------------------
# To forecast future values, we use iterative (recursive) forecasting.
# First, obtain the last available observation from the training data.
last_row <- tail(df_svr, 1)

# For forecasting the next quarter, our predictors will be:
#   lag1 = most recent actual GDPGrowth,
#   lag2 = previous lag1,
#   lag3 = previous lag2,
#   lag4 = previous lag3.
# (Note: In our data, each row already contains GDPGrowth, lag1, lag2, lag3, and lag4.)
# We set up the predictor vector for the next forecast as follows:
current_lags <- c(last_row$GDPGrowth, last_row$lag1, last_row$lag2, last_row$lag3)

# Define the forecast horizon (4 quarters for 2025)
forecast_horizon <- 4
svr_forecasts <- numeric(forecast_horizon)

# Iteratively forecast one quarter ahead at a time.
for(t in 1:forecast_horizon) {
  # Create a one-row data frame with the current lagged values.
  newdata <- data.frame(lag1 = current_lags[1],
                        lag2 = current_lags[2],
                        lag3 = current_lags[3],
                        lag4 = current_lags[4])
  # Predict the next GDPGrowth value.
  pred <- predict(svr_model, newdata = newdata)
  svr_forecasts[t] <- pred
  # Update the lag vector: the new forecast becomes the new lag1,
  # shift the previous values: new lag2 = old lag1, new lag3 = old lag2, new lag4 = old lag3.
  current_lags <- c(pred, current_lags[1:3])
}

# ---------------------------
# 4. Create a Time Index for 2025 Forecasts
# ---------------------------
# Convert the last date to a yearqtr object.
last_yearqtr <- as.yearqtr(last_row$date)

# Create a sequence of 4 quarters starting with the next quarter.
forecast_quarters <- seq(last_yearqtr + 0.25, by = 0.25, length.out = forecast_horizon)

# Convert the yearqtr objects to Date (by default, as.Date returns the first day of the quarter).
forecast_dates <- as.Date(forecast_quarters)

# Create an xts object for the SVR forecasts.
svr_forecast_xts <- xts(svr_forecasts, order.by = forecast_dates)

# ---------------------------
# 5. Plot the SVR Forecast for 2025
# ---------------------------
plot(index(svr_forecast_xts), coredata(svr_forecast_xts), type = "l", col = "blue", lwd = 2,
     xlab = "Date", ylab = "GDP Growth", main = "SVR Forecast for GDP Growth (2025)")
points(index(svr_forecast_xts), coredata(svr_forecast_xts), col = "blue", pch = 16)
